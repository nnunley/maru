while true
    var currentTime := getMilliseconds()
    registerEvents()
    var evaluator := Evaluator.new()
    window.withAllBoxesDo((box) ->
        evaluator.addStreamsFrom(box))
    evaluator.sortAndEvaluateAt(
                window.mapTime(currentTime))
    // layout phase
    window.withAllBoxesDo((box) ->
        box.layOut())
    window.draw()
    sleepUntilNextFrame()
